   414→    /**
   415→     * Start download menggunakan Android DownloadManager
   416→     * - File disimpan di app-specific external storage (TIDAK butuh permission)
   417→     * - Lokasi: Android/data/com.stbmodstore.app/files/Download/STB Mod Store/
   418→     * - Auto-install via FileProvider setelah selesai
   419→     */
   420→    @SuppressLint("Range")
   421→    private void startDownloadWithManager(String url, String userAgent, String contentDisposition, String mimetype, long contentLength) {
   422→        try {
   423→            DownloadManager.Request request = new DownloadManager.Request(Uri.parse(url));
   424→            request.setMimeType(mimetype != null ? mimetype : "application/octet-stream");
   425→            request.setTitle("STB Mod Store - Download APK");
   426→            request.setDescription("Sedang mengunduh APK...");
   427→            request.setAllowedNetworkTypes(DownloadManager.Request.NETWORK_WIFI | DownloadManager.Request.NETWORK_MOBILE);
   428→            request.setAllowedOverRoaming(true);
   429→            request.setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE);
   430→
   431→            // Cookies (untuk auth admin)
   432→            String cookies = CookieManager.getInstance().getCookie(url);
   433→            if (cookies != null) {
   434→                request.addRequestHeader("Cookie", cookies);
   435→            }
   436→            if (userAgent != null) {
   437→                request.addRequestHeader("User-Agent", userAgent);
   438→            }
   439→
   440→            // Tentukan filename
   441→            String fileName = URLUtil.guessFileName(url, contentDisposition, mimetype);
   442→            if (!fileName.toLowerCase().endsWith(".apk") && !fileName.toLowerCase().endsWith(".zip") && !fileName.toLowerCase().endsWith(".rar")) {
   443→                fileName += ".apk";
   444→            }
   445→
   446→            // === SIMPAN DI APP-SPECIFIC STORAGE (TIDAK BUTUH PERMISSION) ===
   447→            // Lokasi: /Android/data/com.stbmodstore.app/files/Download/STB Mod Store/
   448→            File destDir = getExternalFilesDir(Environment.DIRECTORY_DOWNLOADS);
   449→            if (destDir == null) {
   450→                destDir = getFilesDir(); // fallback ke internal storage
   451→            }
   452→            File stbDir = new File(destDir, DOWNLOAD_SUBDIR);
   453→            if (!stbDir.exists()) {
   454→                boolean created = stbDir.mkdirs();
   455→                if (!created && !stbDir.exists()) {
   456→                    stbDir = destDir; // fallback ke parent dir
   457→                }
   458→            }
   459→            File destFile = new File(stbDir, fileName);
   460→
   461→            // Hapus file lama dengan nama yang sama (kalau ada)
   462→            if (destFile.exists()) {
   463→                destFile.delete();
   464→            }
   465→
   466→            // Set destination via URI (file://)
   467→            request.setDestinationUri(Uri.fromFile(destFile));
   468→
   469→            // Enqueue download
   470→            DownloadManager dm = (DownloadManager) getSystemService(Context.DOWNLOAD_SERVICE);
   471→            currentDownloadId = dm.enqueue(request);
   472→
   473→            // Tampilkan progress UI
   474→            long totalBytes = contentLength > 0 ? contentLength : 0;
   475→            showDownloadProgress(fileName, totalBytes);
   476→
   477→            Toast.makeText(this, "Memulai download: " + fileName, Toast.LENGTH_SHORT).show();
   478→
   479→        } catch (SecurityException e) {
   480→            Toast.makeText(this, "Gagal download (permission denied): " + e.getMessage(), Toast.LENGTH_LONG).show();
   481→            e.printStackTrace();
   482→        } catch (Exception e) {
   483→            Toast.makeText(this, "Gagal memulai download: " + e.getMessage(), Toast.LENGTH_LONG).show();
   484→            e.printStackTrace();
   485→        }
   486→    }

   488→    /**
   489→     * Tampilkan progress bar download di bawah aplikasi
   490→     */
   491→    private void showDownloadProgress(String fileName, long totalBytes) {
   492→        runOnUiThread(() -> {
   493→            downloadProgressText.setText("Mengunduh: " + fileName);
   494→            downloadProgressText.setTextColor(Color.parseColor("#065F46"));
   495→            downloadProgressBar.setProgress(0);
   496→            downloadProgressBar.setIndeterminate(false);
   497→            if (totalBytes > 0) {
   498→                downloadProgressPercent.setText("0% • 0 MB / " + formatBytes(totalBytes));
   499→            } else {
   500→                downloadProgressPercent.setText("0% • Mengunduh...");
   501→            }
   502→            downloadProgressContainer.setVisibility(View.VISIBLE);
   503→        });
   504→
   505→        downloadProgressHandler = new Handler(Looper.getMainLooper());
   506→        downloadProgressRunnable = new Runnable() {
   507→            @Override
   508→            public void run() {
   509→                if (currentDownloadId == -1) return;
   510→                DownloadManager.Query query = new DownloadManager.Query();
   511→                query.setFilterById(currentDownloadId);
   512→                DownloadManager dm = (DownloadManager) getSystemService(Context.DOWNLOAD_SERVICE);
   513→                Cursor cursor = dm.query(query);
   514→                if (cursor != null && cursor.moveToFirst()) {
   515→                    int status = cursor.getInt(cursor.getColumnIndex(DownloadManager.COLUMN_STATUS));
   516→                    if (status == DownloadManager.STATUS_RUNNING) {
   517→                        int downloaded = cursor.getInt(cursor.getColumnIndex(DownloadManager.COLUMN_BYTES_DOWNLOADED_SO_FAR));
   518→                        int total = cursor.getInt(cursor.getColumnIndex(DownloadManager.COLUMN_TOTAL_SIZE_BYTES));
   519→                        if (total > 0) {
   520→                            int percent = (downloaded * 100) / total;
   521→                            downloadProgressBar.setProgress(percent);
   522→                            downloadProgressPercent.setText(
   523→                                percent + "% • " + formatBytes(downloaded) + " / " + formatBytes(total)
   524→                            );
   525→                        } else {
   526→                            downloadProgressBar.setIndeterminate(true);
   527→                            downloadProgressPercent.setText(formatBytes(downloaded) + " diunduh...");
   528→                        }
   529→                    }
   530→                    cursor.close();
   531→                }
   532→                downloadProgressHandler.postDelayed(this, 500);
   533→            }
   534→        };
   535→        downloadProgressHandler.post(downloadProgressRunnable);
   536→    }

   538→    private String formatBytes(long bytes) {
   539→        if (bytes < 1024) return bytes + " B";
   540→        if (bytes < 1024 * 1024) return String.format("%.1f KB", bytes / 1024.0);
   541→        if (bytes < 1024L * 1024 * 1024) return String.format("%.1f MB", bytes / (1024.0 * 1024));
   542→        return String.format("%.2f GB", bytes / (1024.0 * 1024 * 1024));
   543→    }

   545→    /**
   546→     * BroadcastReceiver untuk handle download selesai
   547→     */
   548→    private BroadcastReceiver onDownloadComplete = new BroadcastReceiver() {
   549→        @Override
   550→        public void onReceive(Context context, Intent intent) {
   551→            long id = intent.getLongExtra(DownloadManager.EXTRA_DOWNLOAD_ID, -1);
   552→            if (id != currentDownloadId) return;
   553→
   554→            if (downloadProgressHandler != null && downloadProgressRunnable != null) {
   555→                downloadProgressHandler.removeCallbacks(downloadProgressRunnable);
   556→            }
   557→
   558→            // Query status download
   559→            DownloadManager.Query query = new DownloadManager.Query();
   560→            query.setFilterById(id);
   561→            DownloadManager dm = (DownloadManager) getSystemService(Context.DOWNLOAD_SERVICE);
   562→            Cursor cursor = dm.query(query);
   563→
   564→            if (cursor != null && cursor.moveToFirst()) {
   565→                @SuppressLint("Range")
   566→                int status = cursor.getInt(cursor.getColumnIndex(DownloadManager.COLUMN_STATUS));
   567→                @SuppressLint("Range")
   568→                String localUriStr = cursor.getString(cursor.getColumnIndex(DownloadManager.COLUMN_LOCAL_URI));
   569→
   570→                if (status == DownloadManager.STATUS_SUCCESSFUL && localUriStr != null) {
   571→                    // Set progress ke 100%
   572→                    downloadProgressBar.setProgress(100);
   573→                    downloadProgressBar.setIndeterminate(false);
   574→                    downloadProgressText.setText("Download selesai ✓");
   575→                    downloadProgressText.setTextColor(Color.parseColor("#059669"));
   576→                    downloadProgressPercent.setText("Lokasi: Android/data/com.stbmodstore.app/files/Download/STB Mod Store/");
   577→
   578→                    // === AUTO-INSTALL APK VIA FILEPROVIDER ===
   579→                    try {
   580→                        // Parse local file path dari URI
   581→                        Uri localUri = Uri.parse(localUriStr);
   582→                        File apkFile = new File(localUri.getPath());
   583→
   584→                        if (apkFile.exists()) {
   585→                            // Pakai FileProvider untuk install (anti FileUriExposedException di Android 7+)
   586→                            Uri apkContentUri = FileProvider.getUriForFile(
   587→                                context,
   588→                                FILE_PROVIDER_AUTHORITY,
   589→                                apkFile
   590→                            );
   591→
   592→                            Intent installIntent = new Intent(Intent.ACTION_VIEW);
   593→                            installIntent.setDataAndType(apkContentUri, "application/vnd.android.package-archive");
   594→                            installIntent.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
   595→                            installIntent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
   596→                            installIntent.addFlags(Intent.FLAG_ACTIVITY_CLEAR_TOP);
   597→
   598→                            context.startActivity(installIntent);
   599→                            Toast.makeText(context, "APK siap diinstall. Klik Install.", Toast.LENGTH_LONG).show();
   600→                        } else {
   601→                            Toast.makeText(context, "File tidak ditemukan: " + apkFile.getAbsolutePath(), Toast.LENGTH_LONG).show();
   602→                        }
   603→                    } catch (Exception e) {
   604→                        // Fallback: kasih tau user lokasi file
   605→                        Toast.makeText(context,
   606→                            "Download selesai. Buka File Manager → Android/data/com.stbmodstore.app/files/Download/STB Mod Store/",
   607→                            Toast.LENGTH_LONG).show();
   608→                        e.printStackTrace();
   609→                    }
   610→
   611→                    // Sembunyikan progress setelah 8 detik
   612→                    new Handler(Looper.getMainLooper()).postDelayed(() -> {
   613→                        downloadProgressContainer.setVisibility(View.GONE);
   614→                        downloadProgressText.setTextColor(Color.parseColor("#065F46"));
   615→                    }, 8000);
   616→
   617→                } else if (status == DownloadManager.STATUS_FAILED) {
   618→                    @SuppressLint("Range")
   619→                    String reason = cursor.getString(cursor.getColumnIndex(DownloadManager.COLUMN_REASON));
   620→                    downloadProgressText.setText("Download gagal ✗");
   621→                    downloadProgressText.setTextColor(Color.parseColor("#EF4444"));
   622→                    downloadProgressPercent.setText("Alasan: " + (reason != null ? reason : "unknown error"));
   623→
   624→                    Toast.makeText(context,
   625→                        "Download gagal. Coba lagi. (Reason: " + reason + ")",
   626→                        Toast.LENGTH_LONG
   627→                    ).show();
   628→
   629→                    new Handler(Looper.getMainLooper()).postDelayed(() -> {
   630→                        downloadProgressContainer.setVisibility(View.GONE);
   631→                        downloadProgressText.setTextColor(Color.parseColor("#065F46"));
   632→                    }, 5000);
   633→                }
   634→                cursor.close();
   635→            }
   636→            currentDownloadId = -1;
   637→        }
   638→    };

   640→    private boolean isOnline() {
   641→        ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   642→        if (cm == null) return false;
   643→        NetworkInfo info = cm.getActiveNetworkInfo();
   644→        return info != null && info.isConnected();
   645→    }

   675→            downloadProgressHandler.removeCallbacks(downloadProgressRunnable);
   676→        }
   677→        try {
   678→            unregisterReceiver(onDownloadComplete);
   679→        } catch (Exception e) {
   680→            // ignore if not registered
   681→        }
   682→        if (webView != null) {
   683→            webView.destroy();
   684→        }
